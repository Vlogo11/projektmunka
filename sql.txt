CREATE DATABASE autoberles;
USE autoberles;


-- =========================================
-- FELHASZNÁLÓK
-- =========================================

CREATE TABLE felhasznalok (
    felhasznalo_id INT AUTO_INCREMENT PRIMARY KEY,
    nev VARCHAR(100) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    telefonszam VARCHAR(30),
    jelszo_hash VARCHAR(255) NOT NULL,
    szul_datum DATE,
    cim VARCHAR(255),
    profil_tipus ENUM('ugyfel', 'admin') NOT NULL DEFAULT 'ugyfel'
);


-- =========================================
-- JÁRMŰKATEGÓRIÁK
-- =========================================

CREATE TABLE jarmu_kategoriak (
    kategoria_id INT AUTO_INCREMENT PRIMARY KEY,
    nev VARCHAR(100) NOT NULL UNIQUE,
    leiras TEXT
);


-- =========================================
-- JÁRMŰVEK
-- =========================================

CREATE TABLE jarmuvek (
    jarmu_id INT AUTO_INCREMENT PRIMARY KEY,
    kategoria_id INT NOT NULL,
    marka VARCHAR(100) NOT NULL,
    modell VARCHAR(100) NOT NULL,
    alvazszam VARCHAR(50) NOT NULL UNIQUE,
    evjarat YEAR NOT NULL,
    uzemanyag ENUM(
        'benzin',
        'dizel',
        'hibrid',
        'elektromos',
        'egyeb'
    ) NOT NULL,
    valto ENUM(
        'manualis',
        'automata'
    ) NOT NULL,
    szin VARCHAR(50),
    kilometerora_allas INT NOT NULL DEFAULT 0,
    berleti_dij_naponta DECIMAL(10,2),
    veteli_ar DECIMAL(12,2),
    allapot VARCHAR(100),
    elerheto BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT fk_jarmu_kategoria
        FOREIGN KEY (kategoria_id)
        REFERENCES jarmu_kategoriak(kategoria_id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT chk_kilometer
        CHECK (kilometerora_allas >= 0),

    CONSTRAINT chk_berleti_dij
        CHECK (berleti_dij_naponta IS NULL OR berleti_dij_naponta >= 0),

    CONSTRAINT chk_veteli_ar
        CHECK (veteli_ar IS NULL OR veteli_ar >= 0)
);


-- =========================================
-- VÉLEMÉNYEK
-- =========================================

CREATE TABLE velemenyek (
    velemeny_id INT AUTO_INCREMENT PRIMARY KEY,
    felhasznalo_id INT NOT NULL,
    jarmu_id INT NOT NULL,
    ertekeles TINYINT NOT NULL,
    cim VARCHAR(150),
    szoveg TEXT NOT NULL,
    letrehozas_datum DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_velemeny_felhasznalo
        FOREIGN KEY (felhasznalo_id)
        REFERENCES felhasznalok(felhasznalo_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    CONSTRAINT fk_velemeny_jarmu
        FOREIGN KEY (jarmu_id)
        REFERENCES jarmuvek(jarmu_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    CONSTRAINT chk_ertekeles
        CHECK (ertekeles BETWEEN 1 AND 5),

    -- Egy felhasználó egy járműről egyszer írhat véleményt
    CONSTRAINT uq_felhasznalo_jarmu_velemeny
        UNIQUE (felhasznalo_id, jarmu_id)
);


-- =========================================
-- BÉRLÉSEK
-- =========================================

CREATE TABLE berlesek (
    berles_id INT AUTO_INCREMENT PRIMARY KEY,
    felhasznalo_id INT NOT NULL,
    jarmu_id INT NOT NULL,
    kezdet_datum DATETIME NOT NULL,
    veg_datum DATETIME NOT NULL,
    felvetel_helye VARCHAR(255) NOT NULL,
    leadas_helye VARCHAR(255) NOT NULL,
    napi_dij DECIMAL(10,2) NOT NULL,
    osszeg DECIMAL(12,2) NOT NULL,
    fizetesi_mod ENUM(
        'bankkartya',
        'keszpenz',
        'atutalas'
    ) NOT NULL,
    fizetesi_statusz ENUM(
        'fizetetlen',
        'fizetve',
        'visszateritve'
    ) NOT NULL DEFAULT 'fizetetlen',
    berles_statusz ENUM(
        'foglalt',
        'folyamatban',
        'lezart',
        'lemondva'
    ) NOT NULL DEFAULT 'foglalt',
    letrehozas_datum DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_berles_felhasznalo
        FOREIGN KEY (felhasznalo_id)
        REFERENCES felhasznalok(felhasznalo_id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_berles_jarmu
        FOREIGN KEY (jarmu_id)
        REFERENCES jarmuvek(jarmu_id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT chk_berles_datum
        CHECK (veg_datum > kezdet_datum),

    CONSTRAINT chk_napi_dij
        CHECK (napi_dij >= 0),

    CONSTRAINT chk_osszeg
        CHECK (osszeg >= 0)
);


-- =========================================
-- VÁSÁRLÁSOK
-- =========================================

CREATE TABLE vasarlasok (
    vasarlas_id INT AUTO_INCREMENT PRIMARY KEY,
    felhasznalo_id INT NOT NULL,
    jarmu_id INT NOT NULL,
    vasarlas_datum DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    vetel_ar DECIMAL(12,2) NOT NULL,
    fizetesi_mod ENUM(
        'bankkartya',
        'keszpenz',
        'atutalas'
    ) NOT NULL,
    fizetesi_statusz ENUM(
        'fizetetlen',
        'fizetve',
        'visszateritve'
    ) NOT NULL DEFAULT 'fizetetlen',
    vasarlas_statusz ENUM(
        'feldolgozas_alatt',
        'fizetve',
        'teljesitve',
        'lemondva'
    ) NOT NULL DEFAULT 'feldolgozas_alatt',
    atadas_datum DATETIME,

    CONSTRAINT fk_vasarlas_felhasznalo
        FOREIGN KEY (felhasznalo_id)
        REFERENCES felhasznalok(felhasznalo_id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_vasarlas_jarmu
        FOREIGN KEY (jarmu_id)
        REFERENCES jarmuvek(jarmu_id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT chk_vetel_ar
        CHECK (vetel_ar >= 0)
);
